1. FIX: Frontend NOT Calling Price List API

Problem:
Price list endpoint is never triggered.

---

In POS pricing logic (where product price is resolved BEFORE adding to cart):

ADD logic:

```js
if (priceTierEnabled && selectedCustomer) {
  const priceListId = getPriceListIdFromCustomer(selectedCustomer);

  if (priceListId && variant?.bigcommerce_variant_id) {
    const variantIds = [variant.bigcommerce_variant_id];

    console.log("CALLING PRICE LIST API", {
      priceListId,
      variantIds
    });

    const res = await fetch(
      `/api/bigcommerce/price-list/${priceListId}/records?variantIds=${variantIds.join(",")}`
    );

    const priceMap = await res.json();

    console.log("PRICE LIST RESPONSE", priceMap);

    if (priceMap && priceMap[variant.bigcommerce_variant_id]) {
      finalPrice = Number(priceMap[variant.bigcommerce_variant_id]);
      price_source = "price_list";
    }
  }
}
```
