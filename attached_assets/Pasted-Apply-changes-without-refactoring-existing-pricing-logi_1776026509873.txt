5. SYNC FUNCTION (POSTGRES → LOCAL ONLY)

const syncPriceHistory = async () => {
  try {
    setIsSyncing(true);

    const res = await fetch("/api/price-history/sync", {
      headers: {
        "x-user-id": currentUser.id
      }
    });

    const data = await res.json();

    // overwrite local cache completely
    setLocalPriceHistory(data);

    console.log("SYNC COMPLETE", data.length);

  } catch (err) {
    console.error("SYNC FAILED", err);
  } finally {
    setIsSyncing(false);
    setLastSyncTime(Date.now());
  }
};
