```js
const handlePrintInvoice = (order) => {
  const orderId = order.bigcommerce_order_id || order.id;

  if (!orderId) {
    console.warn("Missing order ID for invoice");
    return;
  }

  const url = buildInvoiceUrl(orderId);

  console.log("PRINT INVOICE", {
    orderId,
    url
  });

  // Reuse modal if available

  setShowInvoice(true);

  // Fallback if needed

};
```
