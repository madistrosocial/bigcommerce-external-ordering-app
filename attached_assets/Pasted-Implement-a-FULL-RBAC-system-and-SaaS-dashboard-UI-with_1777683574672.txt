export function hasPermission(user, module, action) {
  if (module === "admin") {
    return user.permissions?.includes("admin:view");
  }

  if (user.user_permissions?.length) {
    return user.user_permissions.includes(`${module}:${action}`);
  }

  if (user.role_permissions?.length) {
    return user.role_permissions.includes(`${module}:${action}`);
  }

  return true; // default allow
}
